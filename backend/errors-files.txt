import { pool } from '../config/db.js';

interface UserProfile {
  domain_scores: Record<string, number>;
  games_played: string[];
  games_weak: string[];
  games_strong: string[];
  last_session_at: string | null;
  days_since_last: number | null;
}

interface Recommendation {
  game_id: string;
  game_name: string;
  domain_id: string;
  reason: string;
  priority: number;
  suggested_mode: string;
}

export async function buildUserProfile(userId: string): Promise<UserProfile> {
  const domainScores: Record<string, number> = {};
  const domains = await pool.query('SELECT id FROM domains');
  for (const d of domains.rows) domainScores[d.id] = 0;

  const gamesAgg = await pool.query<{ game_id: string; n: number; avg_score: number; last_at: string }>(
    `SELECT game_id, COUNT(*)::int AS n, AVG(primary_score)::float AS avg_score, MAX(ended_at) AS last_at
     FROM sessions
     WHERE user_id = $1 AND status = 'accepted'
     GROUP BY game_id`,
    [userId]
  );

  const games_played: string[] = [];
  const games_weak: string[] = [];
  const games_strong: string[] = [];
  let lastSessionAt: string | null = null;

  for (const row of gamesAgg.rows) {
    games_played.push(row.game_id);
    if (row.avg_score < 60) games_weak.push(row.game_id);
    if (row.avg_score >= 80) games_strong.push(row.game_id);
    if (!lastSessionAt || new Date(row.last_at) > new Date(lastSessionAt)) {
      lastSessionAt = row.last_at;
    }
  }

  // Domain scores = avg of game scores per domain
  const domainGames = await pool.query<{ game_id: string; domain_id: string; avg: number }>(
    `SELECT s.game_id, g.domain_id, AVG(s.primary_score)::float AS avg
     FROM sessions s
     JOIN games g ON g.id = s.game_id
     WHERE s.user_id = $1 AND s.status = 'accepted'
     GROUP BY s.game_id, g.domain_id`,
    [userId]
  );

  const domainAgg: Record<string, { sum: number; count: number }> = {};
  for (const row of domainGames.rows) {
    if (!domainAgg[row.domain_id]) domainAgg[row.domain_id] = { sum: 0, count: 0 };
    domainAgg[row.domain_id].sum += row.avg;
    domainAgg[row.domain_id].count += 1;
  }
  for (const [domainId, agg] of Object.entries(domainAgg)) {
    domainScores[domainId] = Math.round((agg.sum / agg.count) * 100) / 100;
  }

  const days = lastSessionAt
    ? Math.floor((Date.now() - new Date(lastSessionAt).getTime()) / (1000 * 60 * 60 * 24))
    : null;

  return {
    domain_scores: domainScores,
    games_played,
    games_weak,
    games_strong,
    last_session_at: lastSessionAt,
    days_since_last: days,
  };
}

export async function getRecommendations(userId: string, limit = 3): Promise<Recommendation[]> {
  const profile = await buildUserProfile(userId);
  const recs: Recommendation[] = [];

  /* Rule 1: User has no sessions — recommend best "starter" games */
  if (profile.games_played.length === 0) {
    const starters = await pool.query<{ id: string; name: string; domain_id: string }>(
      `SELECT id, name, domain_id FROM games
       WHERE id IN ('seq-mem', 'reaction', 'n-back')
       ORDER BY CASE id WHEN 'seq-mem' THEN 1 WHEN 'reaction' THEN 2 ELSE 3 END
       LIMIT $1`,
      [limit]
    );
    return starters.rows.map((g) => ({
      game_id: g.id,
      game_name: g.name,
      domain_id: g.domain_id,
      reason: 'Great starter game — perfect for your first session',
      priority: 100,
      suggested_mode: 'classic',
    }));
  }

  /* Rule 2: If user hasn't played in 3+ days — recommend Daily mode */
  if (profile.days_since_last !== null && profile.days_since_last >= 3) {
    const dailyGame = profile.games_played[0] ?? 'seq-mem';
    const g = await pool.query<{ id: string; name: string; domain_id: string }>(
      'SELECT id, name, domain_id FROM games WHERE id = $1',
      [dailyGame]
    );
    if (g.rowCount) {
      recs.push({
        game_id: g.rows[0].id,
        game_name: g.rows[0].name,
        domain_id: g.rows[0].domain_id,
        reason: `You haven't played in ${profile.days_since_last} days — Daily mode gets you back on track`,
        priority: 95,
        suggested_mode: 'daily',
      });
    }
  }

  /* Rule 3: Weak games — recommend more practice */
  const weakToRecommend = profile.games_weak
    .filter((id) => !recs.find((r) => r.game_id === id))
    .slice(0, 1);
  for (const gameId of weakToRecommend) {
    const g = await pool.query<{ id: string; name: string; domain_id: string }>(
      'SELECT id, name, domain_id FROM games WHERE id = $1',
      [gameId]
    );
    if (g.rowCount) {
      recs.push({
        game_id: g.rows[0].id,
        game_name: g.rows[0].name,
        domain_id: g.rows[0].domain_id,
        reason: 'Your accuracy is below 60% — try Precision mode to build consistency',
        priority: 80,
        suggested_mode: 'precision',
      });
    }
  }

  /* Rule 4: Weakest domain — recommend a game from it */
  const weakest = Object.entries(profile.domain_scores)
    .filter(([id, score]) => score > 0 && score < 70 && !recs.find((r) => r.domain_id === id))
    .sort((a, b) => a[1] - b[1])[0];

  if (weakest && recs.length < limit) {
    const g = await pool.query<{ id: string; name: string; domain_id: string }>(
      `SELECT id, name, domain_id FROM games
       WHERE domain_id = $1
       AND id <> ALL($2::text[])
       ORDER BY RANDOM() LIMIT 1`,
      [weakest[0], profile.games_played]
    );
    if (g.rowCount) {
      recs.push({
        game_id: g.rows[0].id,
        game_name: g.rows[0].name,
        domain_id: g.rows[0].domain_id,
        reason: `Your ${weakest[0]} domain needs attention (score: ${weakest[1]})`,
        priority: 75,
        suggested_mode: 'classic',
      });
    }
  }

  /* Rule 5: Challenge a strength — push to next level */
  if (profile.games_strong.length > 0 && recs.length < limit) {
    const strongGame = profile.games_strong[0]!;
    const already = recs.find((r) => r.game_id === strongGame);
    if (!already) {
      const g = await pool.query<{ id: string; name: string; domain_id: string }>(
        'SELECT id, name, domain_id FROM games WHERE id = $1',
        [strongGame]
      );
      if (g.rowCount) {
        recs.push({
          game_id: g.rows[0].id,
          game_name: g.rows[0].name,
          domain_id: g.rows[0].domain_id,
          reason: `You excel here — try Endless mode to test your limits`,
          priority: 70,
          suggested_mode: 'endless',
        });
      }
    }
  }

  /* Rule 6: Fill remaining slots with popular games user hasn't played */
  if (recs.length < limit) {
    const remaining = limit - recs.length;
    const existing = recs.map((r) => r.game_id);
    const allExcluded = [...existing, ...profile.games_played];
    const fillers = await pool.query<{ id: string; name: string; domain_id: string }>(
      `SELECT id, name, domain_id FROM games
       WHERE status = 'ready'
       AND id <> ALL($1::text[])
       ORDER BY RANDOM() LIMIT $2`,
      [allExcluded, remaining]
    );
    for (const g of fillers.rows) {
      recs.push({
        game_id: g.id,
        game_name: g.name,
        domain_id: g.domain_id,
        reason: 'Try something new to broaden your cognitive profile',
        priority: 50,
        suggested_mode: 'classic',
      });
    }
  }

  recs.sort((a, b) => b.priority - a.priority);
  return recs.slice(0, limit);
}

import { pool } from '../../config/db.js';

export interface DailyPlanItem {
  game_id: string;
  game_name: string;
  domain_id: string;
  mode: string;
  minutes: number;
  reason: string;
}

export interface DailyPlanResult {
  plan: DailyPlanItem[];
  summary: string;
  generated_at: string;
  source: 'llm' | 'rule' | 'cache';
  language: Lang;
  model: string;
  latency_ms: number;
  tokens?: { prompt: number; completion: number; total: number };
  validation_issues?: { code: string; detail: string }[];
}

/* Simple in-memory cache keyed by userId:YYYY-MM-DD */
const memCache = new Map<string, DailyPlanResult>();
const CACHE_TTL_MS = 6 * 60 * 60 * 1000; // 6 hours
const cacheTimestamps = new Map<string, number>();

function cacheKey(userId: string, lang: Lang): string {
  const d = new Date().toISOString().slice(0, 10);
  return `${userId}:${d}:${lang}`;
}

function getCached(userId: string, lang: Lang): DailyPlanResult | null {
  const k = cacheKey(userId, lang);
  const ts = cacheTimestamps.get(k);
  if (ts && Date.now() - ts < CACHE_TTL_MS) {
    const hit = memCache.get(k);
    if (hit) return { ...hit, source: 'cache' };
  }
  return null;
}

function setCached(userId: string, lang: Lang, result: DailyPlanResult): void {
  const k = cacheKey(userId, lang);
  memCache.set(k, result);
  cacheTimestamps.set(k, Date.now());
}

/* ─── Robust JSON extraction ─── */
function extractJson(text: string): { summary?: unknown; games?: unknown } | null {
  const candidates: string[] = [];
  candidates.push(text.trim());
  const fenceMatch = text.match(/```(?:json)?\s*([\s\S]*?)```/i);
  if (fenceMatch) candidates.push(fenceMatch[1].trim());
  const first = text.indexOf('{');
  const last = text.lastIndexOf('}');
  if (first !== -1 && last > first) {
    candidates.push(text.slice(first, last + 1).trim());
  }
  for (const c of candidates) {
    if (!c) continue;
    try {
      const obj = JSON.parse(c);
      if (obj && typeof obj === 'object' && !Array.isArray(obj)) {
        return obj as { summary?: unknown; games?: unknown };
      }
    } catch { /* try next */ }
  }
  return null;
}
/* ─── Ensure reasons are varied + reference data ─── */
function enrichReasons(
  plan: DailyPlanItem[],
  candidates: DailyPlanCandidate[],
  language: Lang
): DailyPlanItem[] {
  const candMap = new Map(candidates.map((c) => [c.game_id, c]));
  const seen = new Set<string>();
  const result: DailyPlanItem[] = [];

  for (let i = 0; i < plan.length; i++) {
    const item = plan[i];
    const cand = candMap.get(item.game_id);
    let reason = item.reason.trim();

    // Detect duplication or generic reasons
    const isDuplicate = seen.has(reason.toLowerCase());
    const isTooGeneric =
      reason.length < 15 ||
      /^(لتنويع|لتحسين|لتطوير|to improve|for variety|good for)/i.test(reason) && reason.length < 40;

    if (isDuplicate || isTooGeneric || !cand) {
      // Generate a varied reason from candidate data
      const fallbackReason = buildVariedReason(cand, i, language);
      reason = fallbackReason;
    }
    seen.add(reason.toLowerCase());

    result.push({ ...item, reason });
  }
  return result;
}

function buildVariedReason(
  cand: DailyPlanCandidate | undefined,
  index: number,
  language: Lang
): string {
  if (!cand) {
    return language === 'ar' ? 'اختار متنوع.' : 'Good for variety.';
  }
  const stage =
    index === 0 ? 'warmup' : index === 1 ? 'main' : index === 2 ? 'main2' : 'finish';
  const level = cand.current_level;

  if (language === 'ar') {
    if (cand.rule_reason.includes('never played')) {
      return index === 0
        ? 'جديد — ابدأ بيه خفيف (3 دقايق).'
        : 'مجال جديد — وسّع تدريبك.';
    }
    if (cand.rule_reason.includes('low accuracy')) {
      return `دقتك هنا ضعيفة — محتاج شغل على مستوى ${level}.`;
    }
    if (cand.rule_reason.includes('not played recently')) {
      return `مش لعبته من فترة — جدد المهارة.`;
    }
    return `ثابت على مستوى ${level} — عزّز الأداء.`;
  } else {
    if (cand.rule_reason.includes('never played')) {
      return index === 0 ? 'New — start light (3 min).' : 'New domain — broaden your range.';
    }
    if (cand.rule_reason.includes('low accuracy')) {
      return `Accuracy gap here — work at level ${level}.`;
    }
    if (cand.rule_reason.includes('not played recently')) {
      return `Haven't played in a while — refresh the skill.`;
    }
    return `Solid at level ${level} — reinforce the pattern.`;
  }
}
export async function generateDailyPlan(
  userId: string,
  language: Lang = 'en'
): Promise<DailyPlanResult> {
  const started = Date.now();

  const cached = getCached(userId, language);
  if (cached) return { ...cached, latency_ms: 0 };

  const [ctx, candidates] = await Promise.all([
    loadContext(userId),
    loadCandidates(userId),
  ]);

  if (candidates.length === 0) {
    return ruleFallback(userId, language, started, 'no_candidates');
  }

  const systemPrompt =
    language === 'ar' ? SYSTEM_PROMPT_DAILY_PLAN_AR : SYSTEM_PROMPT_DAILY_PLAN_EN;
  const userPrompt = buildDailyPlanUserPrompt(ctx, candidates, language);
  const model = language === 'ar' ? env.GROQ_MODEL_AR : env.GROQ_MODEL_EN;

  let llmText = '';
  let tokens: DailyPlanResult['tokens'];
  let usedModel = '';
  let usedProvider = '';

  try {
    const res = await routeChat(
      [
        { role: 'system', content: systemPrompt },
        { role: 'user', content: userPrompt },
      ],
      { temperature: 0.4, max_tokens: 1400, language, json_mode: true }
    );
    llmText = res.content;
    usedModel = res.model;
    usedProvider = res.provider;
    tokens = {
      prompt: res.usage.prompt_tokens,
      completion: res.usage.completion_tokens,
      total: res.usage.total_tokens,
    };
    console.error('[daily-plan] provider=' + usedProvider, '| model=' + usedModel, '| chain=' + res.fallback_chain.join('>'));
  } catch (err) {
    console.error('[daily-plan] all providers failed:', (err as Error).message);
    return ruleFallback(userId, language, started, 'llm_error');
  }

  /* Parse JSON — try multiple strategies */
  const parsed = extractJson(llmText);
  if (!parsed) {
    console.error('[daily-plan] JSON parse failed.');
    console.error('[daily-plan] Raw LLM output (first 800 chars):');
    console.error(llmText.slice(0, 800));
    console.error('[daily-plan] ---');
    return ruleFallback(userId, language, started, 'json_parse_error');
  }

  const summary = typeof parsed.summary === 'string' ? parsed.summary.trim() : '';
  const gamesArr = Array.isArray(parsed.games) ? parsed.games : [];

  if (!summary || gamesArr.length === 0) {
    return ruleFallback(userId, language, started, 'missing_fields');
  }

  /* Validate the summary text */
  const summaryCheck = validateAiText(summary, { minLen: 10, maxLen: 400 });
  if (!summaryCheck.ok) {
    return ruleFallback(userId, language, started, 'summary_invalid');
  }

  /* Build plan items, filtered against candidate whitelist */
  const candMap = new Map(candidates.map((c) => [c.game_id, c]));
  const plan: DailyPlanItem[] = [];
  for (const g of gamesArr) {
    if (!g || typeof g !== 'object') continue;
    const item = g as { game_id?: unknown; reason?: unknown };
    const gid = typeof item.game_id === 'string' ? item.game_id : '';
    const reasonRaw = typeof item.reason === 'string' ? item.reason.trim() : '';
    if (!gid || !reasonRaw) continue;
    const cand = candMap.get(gid);
    if (!cand) continue; // hallucinated game_id — reject silently
    const reasonCheck = validateAiText(reasonRaw, { minLen: 5, maxLen: 200 });
    if (!reasonCheck.ok) continue;
    plan.push({
      game_id: cand.game_id,
      game_name: cand.game_name,
      domain_id: cand.domain_id,
      mode: cand.mode,
      minutes: cand.suggested_minutes,
      reason: reasonCheck.cleaned,
    });
    if (plan.length >= 5) break;
  }

  const enrichedPlan = enrichReasons(plan, candidates, language);
  plan.length = 0;
  plan.push(...enrichedPlan);

  if (plan.length < 2) {
    console.error('[daily-plan] Only', plan.length, 'valid items. LLM games:', JSON.stringify(gamesArr), 'Candidates:', candidates.map(c => c.game_id));
    return ruleFallback(userId, language, started, 'too_few_valid_items');
  }

  const result: DailyPlanResult = {
    plan,
    summary: summaryCheck.cleaned,
    generated_at: new Date().toISOString(),
    source: 'llm',
    language,
    model: usedModel,
    latency_ms: Date.now() - started,
    tokens,
  };

  setCached(userId, language, result);
  return result;
}

/* ─── Context loader ─── */
async function loadContext(userId: string): Promise<DailyPlanContext> {
  /* User's own aggregates */
  const aggRes = await pool.query<{
    total: number;
    sessions_7d: number;
    sessions_today: number;
    avg_acc_7d: number | null;
    last_at: string | null;
  }>(
    `SELECT
       (SELECT COUNT(*)::int FROM sessions WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL) AS total,
       (SELECT COUNT(*)::int FROM sessions WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL AND ended_at >= NOW() - INTERVAL '7 days') AS sessions_7d,
       (SELECT COUNT(*)::int FROM sessions WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL AND ended_at >= CURRENT_DATE) AS sessions_today,
       (SELECT AVG(accuracy)::float FROM sessions WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL AND ended_at >= NOW() - INTERVAL '7 days') AS avg_acc_7d,
       (SELECT MAX(ended_at) FROM sessions WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL) AS last_at`,
    [userId]
  );
  const row = aggRes.rows[0];

  /* Domain strengths/weaknesses */
  const domRes = await pool.query<{ domain_id: string; avg_acc: number }>(
    `SELECT g.domain_id, AVG(s.accuracy)::float AS avg_acc
       FROM sessions s
       JOIN games g ON g.id = s.game_id
      WHERE s.user_id = $1 AND s.status='accepted' AND s.deleted_at IS NULL
        AND s.ended_at >= NOW() - INTERVAL '30 days'
      GROUP BY g.domain_id
      HAVING COUNT(*) >= 2
      ORDER BY avg_acc DESC`,
    [userId]
  );
  const strengths = domRes.rows.slice(0, 3).map((r) => r.domain_id);
  const weaknesses = domRes.rows.slice(-3).reverse().map((r) => r.domain_id);

  /* Recent engine decisions */
  const decRes = await pool.query<{ reason_code: string; n: number }>(
    `SELECT reason_code, COUNT(*)::int AS n
       FROM engine_decisions
      WHERE user_id = $1 AND created_at >= NOW() - INTERVAL '7 days'
      GROUP BY reason_code`,
    [userId]
  );
  let promos = 0;
  let demos = 0;
  for (const r of decRes.rows) {
    if (r.reason_code === 'PROMOTE_STABLE') promos += r.n;
    if (r.reason_code === 'DEMOTE_STRUGGLE') demos += r.n;
  }

  const lastAt = row.last_at ? new Date(row.last_at).getTime() : null;
  const daysSince = lastAt ? Math.floor((Date.now() - lastAt) / 86_400_000) : null;

  const total = row.total ?? 0;
  const tier: DailyPlanContext['evidence_tier'] =
    total < 5 ? 'calibrating' : total < 20 ? 'provisional' : 'established';

  return {
    total_sessions: total,
    sessions_last_7d: row.sessions_7d ?? 0,
    sessions_today: row.sessions_today ?? 0,
    avg_accuracy_last_7d: row.avg_acc_7d ?? 0,
    days_since_last: daysSince,
    current_streak_days: 0,
    top_strengths: strengths,
    top_weaknesses: weaknesses,
    recent_promotions: promos,
    recent_demotions: demos,
    evidence_tier: tier,
  };
}

/* ─── Candidate loader ─── */
async function loadCandidates(userId: string): Promise<DailyPlanCandidate[]> {
  /* Pick top games by weakness + freshness */
  const res = await pool.query<{
    game_id: string;
    game_name: string;
    domain_id: string;
    avg_acc: number | null;
    plays: number;
    last_played: string | null;
    current_level: number | null;
  }>(
    `SELECT g.id AS game_id, g.name AS game_name, g.domain_id,
            AVG(s.accuracy)::float AS avg_acc,
            COUNT(s.id)::int AS plays,
            MAX(s.ended_at) AS last_played,
            MAX(pga.current_level)::int AS current_level
       FROM games g
       LEFT JOIN sessions s
         ON s.game_id = g.id
        AND s.user_id = $1
        AND s.status='accepted'
        AND s.deleted_at IS NULL
        AND s.ended_at >= NOW() - INTERVAL '30 days'
       LEFT JOIN player_game_ability pga
         ON pga.game_id = g.id AND pga.user_id = $1
      WHERE g.status = 'ready'
      GROUP BY g.id, g.name, g.domain_id
      ORDER BY g.domain_id, g.name`,
    [userId]
  );

  const now = Date.now();
  const candidates: DailyPlanCandidate[] = [];

  for (const r of res.rows) {
    /* Skip games played today */
    if (r.last_played && new Date(r.last_played).toDateString() === new Date().toDateString()) {
      continue;
    }
    const avgAcc = r.avg_acc ?? 0;
    const playedRecently = r.last_played
      ? now - new Date(r.last_played).getTime() < 3 * 86_400_000
      : false;
    const level = r.current_level ?? 3;

    let reason: string;
    if (r.plays === 0) {
      reason = 'never played - good for variety';
    } else if (avgAcc < 70) {
      reason = `low accuracy (${Math.round(avgAcc)}%) - needs work`;
    } else if (!playedRecently && r.plays > 0) {
      reason = 'not played recently - keep it fresh';
    } else {
      reason = `solid at level ${level}`;
    }

    candidates.push({
      game_id: r.game_id,
      game_name: r.game_name,
      domain_id: r.domain_id,
      mode: 'classic',
      suggested_minutes: r.plays === 0 ? 3 : 4,
      current_level: level,
      rule_reason: reason,
    });
  }

  /* Sort: never-played first, then low-accuracy, then rest */
  candidates.sort((a, b) => {
    const aNew = a.rule_reason.includes('never played') ? 0 : 1;
    const bNew = b.rule_reason.includes('never played') ? 0 : 1;
    if (aNew !== bNew) return aNew - bNew;
    const aLow = a.rule_reason.includes('low accuracy') ? 0 : 1;
    const bLow = b.rule_reason.includes('low accuracy') ? 0 : 1;
    return aLow - bLow;
  });

  return candidates.slice(0, 10);
}

/* ─── Rule-based fallback (uses existing planner) ─── */
async function ruleFallback(
  userId: string,
  language: Lang,
  startedAt: number,
  reason: string
): Promise<DailyPlanResult> {
  const fallbackPlan: DailyPlanItem[] = [];
  try {
    /* Reuse existing planner via dynamic import to avoid cycle */
    const mod = await import('../planner.js');
    const plan = await mod.buildDailyPlan(userId);
    for (const s of plan.sessions.slice(0, 4)) {
      fallbackPlan.push({
        game_id: s.game_id,
        game_name: s.game_name,
        domain_id: '',
        mode: s.mode,
        minutes: s.estimated_minutes,
        reason: s.reason,
      });
    }
  } catch {
    /* ignore */
  }

  const summary =
    language === 'ar'
      ? 'خطة بسيطة مبنية على بياناتك. ابدأ بجلسة واحدة وقسّم الباقي على اليوم.'
      : 'A simple plan based on your data. Start with one session and spread the rest across the day.';

  return {
    plan: fallbackPlan,
    summary,
    generated_at: new Date().toISOString(),
    source: 'rule',
    language,
    model: 'fallback/planner',
    latency_ms: Date.now() - startedAt,
    validation_issues: reason ? [{ code: reason, detail: 'fell back to rule-based' }] : undefined,
  };
}

export interface FingerprintResult {
  archetype: Archetype;
  archetype_label: string;
  modifiers: Modifier[];
  modifier_labels: string[];
  summary: string;
  traits: {

const MIN_SESSIONS = 3;
const CACHE_TTL_MS = 6 * 60 * 60 * 1000; // 6 hours

/* --- Compute stats from sessions --- */
interface RawStats {
  sessions_count: number;
  avg_accuracy: number;
  median_rt_ms: number;
  rt_std_ms: number;
  accuracy_slope: number;
  unique_games_played: number;
  domains_touched: number;
  top_domain: string | null;
  top_domain_accuracy: number | null;
  weakest_domain: string | null;
  weakest_domain_accuracy: number | null;
  domain_std: number;
}

async function loadRawStats(userId: string): Promise<RawStats> {
  const statsRes = await pool.query<{
    total: number;
    avg_acc: number | null;
    median_rt: number | null;
    rt_std: number | null;
    unique_games: number;
    domains: number;
  }>(
    `SELECT
       COUNT(*)::int AS total,
       AVG(accuracy)::float AS avg_acc,
       PERCENTILE_CONT(0.5) WITHIN GROUP (ORDER BY median_rt_ms)::float AS median_rt,
       STDDEV(median_rt_ms)::float AS rt_std,
       COUNT(DISTINCT game_id)::int AS unique_games,
       COUNT(DISTINCT domain_id)::int AS domains
     FROM sessions
     WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL`,
    [userId]
  );
  const s = statsRes.rows[0];

  /* Domain breakdown */
  const domRes = await pool.query<{ domain_id: string; avg_acc: number; n: number }>(
    `SELECT domain_id, AVG(accuracy)::float AS avg_acc, COUNT(*)::int AS n
     FROM sessions
     WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL
     GROUP BY domain_id
     HAVING COUNT(*) >= 1
     ORDER BY avg_acc DESC`,
    [userId]
  );
  const topDom = domRes.rows[0] ?? null;
  const lastDom = domRes.rows[domRes.rows.length - 1] ?? null;
  const weakDom = topDom && lastDom && topDom.domain_id === lastDom.domain_id ? null : lastDom;

  /* domain std */
  let domainStd = 0;
  if (domRes.rows.length >= 2) {
    const vals = domRes.rows.map((r) => r.avg_acc);
    const mean = vals.reduce((a, b) => a + b, 0) / vals.length;
    const variance = vals.reduce((a, b) => a + (b - mean) ** 2, 0) / vals.length;
    domainStd = Math.sqrt(variance);
  }

  /* accuracy slope: first 5 vs last 5 sessions */
  const slopeRes = await pool.query<{ accuracy: number; ended_at: string }>(
    `SELECT accuracy, ended_at FROM sessions
     WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL
     ORDER BY ended_at ASC`,
    [userId]
  );
  let slope = 0;
  if (slopeRes.rows.length >= 4) {
    const half = Math.floor(slopeRes.rows.length / 2);
    const firstHalf = slopeRes.rows.slice(0, half);
    const lastHalf = slopeRes.rows.slice(-half);
    const avgFirst = firstHalf.reduce((a, r) => a + r.accuracy, 0) / firstHalf.length;
    const avgLast = lastHalf.reduce((a, r) => a + r.accuracy, 0) / lastHalf.length;
    slope = (avgLast - avgFirst) / Math.max(1, half);
  }

  return {
    sessions_count: s.total ?? 0,
    avg_accuracy: s.avg_acc ?? 0,
    median_rt_ms: s.median_rt ?? 0,
    rt_std_ms: s.rt_std ?? 0,
    accuracy_slope: Math.round(slope * 100) / 100,
    unique_games_played: s.unique_games ?? 0,
    domains_touched: s.domains ?? 0,
    top_domain: topDom?.domain_id ?? null,
    top_domain_accuracy: topDom?.avg_acc ?? null,
    weakest_domain: weakDom?.domain_id ?? null,
    weakest_domain_accuracy: weakDom?.avg_acc ?? null,
    domain_std: Math.round(domainStd * 100) / 100,
  };
}

/* --- Pick archetype --- */
function pickArchetype(s: RawStats): Archetype {
  const acc = s.avg_accuracy;
  const rt = s.median_rt_ms;

  /* Priority 1: Specialist (uneven domains) */
  const spread = (s.top_domain_accuracy ?? 0) - (s.weakest_domain_accuracy ?? 0);
  if (s.domains_touched >= 3 && spread >= 20 && s.domain_std >= 8) {
    return 'specialist';
  }

  /* Priority 2: Explorer (broad variety) */
  if (s.unique_games_played >= 20 && s.domains_touched >= 6) {
    return 'explorer';
  }

  /* Priority 3: RT × Accuracy matrix */
  const isFast = rt > 0 && rt < 700;
  const isSlow = rt > 1200;
  const isHighAcc = acc >= 80;
  const isLowAcc = acc < 65;

  if (isFast && isHighAcc) return 'sharp_shooter';
  if (isFast && isLowAcc) return 'impulsive';
  if (isSlow && isHighAcc) return 'methodical';
  if (isSlow && isLowAcc) return 'cautious';
  if (isHighAcc) return 'balanced';
  return 'exploring';
}

/* --- Pick modifiers --- */
function pickModifiers(s: RawStats): Modifier[] {
  const mods: Modifier[] = [];
  if (s.accuracy_slope >= 0.3) mods.push('learner');
  if (Math.abs(s.accuracy_slope) < 0.1 && s.sessions_count >= 10) mods.push('plateaued');
  if (s.rt_std_ms > 400) mods.push('variable');
  if (s.domain_std < 5 && s.sessions_count >= 15) mods.push('specialist');
  if (s.domain_std > 12) mods.push('generalist');
  return mods;
}

/* --- Confidence (0-1) --- */
function computeConfidence(s: RawStats): number {
  const sessionScore = Math.min(1, s.sessions_count / 30);
  const domainScore = Math.min(1, s.domains_touched / 6);
  const gameScore = Math.min(1, s.unique_games_played / 15);
  return Math.round(((sessionScore * 0.5) + (domainScore * 0.3) + (gameScore * 0.2)) * 100) / 100;
}

/* --- Evidence tier --- */
function computeTier(sessions: number): FingerprintFactSheet['evidence_tier'] {
  if (sessions < 5) return 'calibrating';
  if (sessions < 20) return 'provisional';
  return 'established';
}

/* --- Deterministic fallback summary --- */
function buildFallbackSummary(f: FingerprintFactSheet, lang: Lang): string {
  const name = archetypeLabel(f.archetype, lang);
  if (lang === 'ar') {
    return `بصمتك المعرفية: ${name}. متوسط دقتك ${Math.round(f.avg_accuracy)}% ومتوسط زمن ردك ${Math.round(f.median_rt_ms)} مللي ثانية عبر ${f.sessions_used} جلسة. أقوى مجال: ${f.top_domain ?? '-'}، وأضعف مجال: ${f.weakest_domain ?? '-'}.`;
  }
  return `Your cognitive fingerprint: ${name}. Average accuracy ${Math.round(f.avg_accuracy)}% with a median RT of ${Math.round(f.median_rt_ms)}ms across ${f.sessions_used} sessions. Strongest domain: ${f.top_domain ?? '-'}. Weakest: ${f.weakest_domain ?? '-'}.`;
}

export async function generateFingerprint(
  userId: string,
  language: Lang = 'en',
  forceRefresh = false
): Promise<FingerprintResult> {
  const started = Date.now();

  /* 1) Cache check */
  if (!forceRefresh) {
    const cached = await pool.query<{
      archetype: string;
      summary: string;
      traits: Record<string, unknown>;
      confidence: number;
      sessions_used: number;
      generated_at: string;
    }>(
      `SELECT archetype, summary, traits, confidence, sessions_used, generated_at
         FROM cognitive_fingerprints
        WHERE user_id = $1
        ORDER BY generated_at DESC
        LIMIT 1`,
      [userId]
    );
    const row = cached.rows[0];
    const cacheLang = row ? ((row.traits as Record<string, unknown>).language as string | undefined) : undefined;
    if (row && cacheLang === language) {
      const age = Date.now() - new Date(row.generated_at).getTime();
      if (age < CACHE_TTL_MS) {
        const arch = row.archetype as Archetype;
        const mods = (row.traits.modifiers as Modifier[]) ?? [];
        return {
          archetype: arch,
          archetype_label: archetypeLabel(arch, language),
          modifiers: mods,
          modifier_labels: mods.map((m) => modifierLabel(m, language)),
          summary: row.summary,
          traits: {
            avg_accuracy: (row.traits.avg_accuracy as number) ?? 0,
            median_rt_ms: (row.traits.median_rt_ms as number) ?? 0,
            rt_std_ms: (row.traits.rt_std_ms as number) ?? 0,
            accuracy_slope: (row.traits.accuracy_slope as number) ?? 0,
            top_domain: (row.traits.top_domain as string | null) ?? null,
            top_domain_accuracy: (row.traits.top_domain_accuracy as number | null) ?? null,
            weakest_domain: (row.traits.weakest_domain as string | null) ?? null,
            weakest_domain_accuracy: (row.traits.weakest_domain_accuracy as number | null) ?? null,
            unique_games_played: (row.traits.unique_games_played as number) ?? 0,
            domains_touched: (row.traits.domains_touched as number) ?? 0,
          },
          evidence_tier: computeTier(row.sessions_used),
          confidence: row.confidence,
          sessions_used: row.sessions_used,
          source: 'cache',
          language,
          model: 'cache',
          provider: 'none',
          latency_ms: Date.now() - started,
        };
      }
    }
  }

  /* 2) Load raw stats */
  const stats = await loadRawStats(userId);

  if (stats.sessions_count < MIN_SESSIONS) {
    const arch: Archetype = 'exploring';
    return {
      archetype: arch,
      archetype_label: archetypeLabel(arch, language),
      modifiers: [],
      modifier_labels: [],
      summary:
        language === 'ar'
          ? `لسه بدري — محتاجين ${MIN_SESSIONS} جلسات على الأقل لبناء بصمتك المعرفية. لسه عندك ${stats.sessions_count}.`
          : `Still forming — need at least ${MIN_SESSIONS} sessions to build your cognitive fingerprint. You have ${stats.sessions_count}.`,
      traits: {
        avg_accuracy: stats.avg_accuracy,
        median_rt_ms: stats.median_rt_ms,
        rt_std_ms: stats.rt_std_ms,
        accuracy_slope: stats.accuracy_slope,
        top_domain: stats.top_domain,
        top_domain_accuracy: stats.top_domain_accuracy,
        weakest_domain: stats.weakest_domain,
        weakest_domain_accuracy: stats.weakest_domain_accuracy,
        unique_games_played: stats.unique_games_played,
        domains_touched: stats.domains_touched,
      },
      evidence_tier: 'calibrating',
      confidence: computeConfidence(stats),
      sessions_used: stats.sessions_count,
      source: 'insufficient_data',
      language,
      model: 'none',
      provider: 'none',
      latency_ms: Date.now() - started,
      message:
        language === 'ar'
          ? 'لسه بدري — العب شوية كمان'
          : 'Not enough data yet — keep playing',
    };
  }

  /* 3) Classify */
  const archetype = pickArchetype(stats);
  const modifiers = pickModifiers(stats);
  const tier = computeTier(stats.sessions_count);
  const confidence = computeConfidence(stats);

  const sheet: FingerprintFactSheet = {
    archetype,
    modifiers,
    sessions_used: stats.sessions_count,
    evidence_tier: tier,
    avg_accuracy: stats.avg_accuracy,
    median_rt_ms: stats.median_rt_ms,
    rt_std_ms: stats.rt_std_ms,
    accuracy_slope: stats.accuracy_slope,
    top_domain: stats.top_domain,
    top_domain_accuracy: stats.top_domain_accuracy,
    weakest_domain: stats.weakest_domain,
    weakest_domain_accuracy: stats.weakest_domain_accuracy,
    unique_games_played: stats.unique_games_played,
    domains_touched: stats.domains_touched,
  };

  /* 4) LLM phrasing */
  const systemPrompt =
    language === 'ar' ? SYSTEM_PROMPT_FINGERPRINT_AR : SYSTEM_PROMPT_FINGERPRINT_EN;
  const userPrompt = buildFingerprintUserPrompt(sheet, language);

  let summary = '';
  let usedModel = '';
  let usedProvider = '';
  let source: FingerprintResult['source'] = 'llm';

  try {
    const res = await routeChat(
      [
        { role: 'system', content: systemPrompt },
        { role: 'user', content: userPrompt },
      ],
      { temperature: 0.55, max_tokens: 500, language }
    );
    const check = validateAiText(res.content, { minLen: 60, maxLen: 1200 });
    if (check.ok) {
      summary = check.cleaned;
      usedModel = res.model;
      usedProvider = res.provider;
      console.error('[fingerprint] provider=' + usedProvider, 'model=' + usedModel, 'chain=' + res.fallback_chain.join('>'));
    } else {
      summary = buildFallbackSummary(sheet, language);
      source = 'rule';
      usedModel = 'fallback';
      usedProvider = 'none';
    }
  } catch (err) {
    console.error('[fingerprint] all providers failed:', (err as Error).message);
    console.error('[fingerprint] lang=', language);
    summary = buildFallbackSummary(sheet, language);
    source = 'rule';
    usedModel = 'fallback';
    usedProvider = 'none';
  }

  /* 5) Store */
  try {
    await pool.query(
      `INSERT INTO cognitive_fingerprints (user_id, archetype, summary, traits, confidence, sessions_used)
       VALUES ($1, $2, $3, $4, $5, $6)`,
      [
        userId,
        archetype,
        summary,
        JSON.stringify({
          language,
          modifiers,
          avg_accuracy: stats.avg_accuracy,
          median_rt_ms: stats.median_rt_ms,
          rt_std_ms: stats.rt_std_ms,
          accuracy_slope: stats.accuracy_slope,
          top_domain: stats.top_domain,
          top_domain_accuracy: stats.top_domain_accuracy,
          weakest_domain: stats.weakest_domain,
          weakest_domain_accuracy: stats.weakest_domain_accuracy,
          unique_games_played: stats.unique_games_played,
          domains_touched: stats.domains_touched,
        }),
        confidence,
        stats.sessions_count,
      ]
    );
  } catch (err) {
    console.warn('[fingerprint] store failed:', (err as Error).message);
  }

  return {
    archetype,
    archetype_label: archetypeLabel(archetype, language),
    modifiers,
    modifier_labels: modifiers.map((m) => modifierLabel(m, language)),
    summary,
    traits: {
      avg_accuracy: Math.round(stats.avg_accuracy * 10) / 10,
      median_rt_ms: Math.round(stats.median_rt_ms),
      rt_std_ms: Math.round(stats.rt_std_ms),
      accuracy_slope: stats.accuracy_slope,
      top_domain: stats.top_domain,
      top_domain_accuracy: stats.top_domain_accuracy ? Math.round(stats.top_domain_accuracy * 10) / 10 : null,
      weakest_domain: stats.weakest_domain,
      weakest_domain_accuracy: stats.weakest_domain_accuracy ? Math.round(stats.weakest_domain_accuracy * 10) / 10 : null,
      unique_games_played: stats.unique_games_played,
      domains_touched: stats.domains_touched,
    },
    evidence_tier: tier,
    confidence,
    sessions_used: stats.sessions_count,
    source,
    language,
    model: usedModel,
    provider: usedProvider,
    latency_ms: Date.now() - started,
  };
}

async function getUserSessionCount(userId: string): Promise<number> {
  const r = await pool.query<{ n: number }>(
    `SELECT COUNT(*)::int AS n FROM sessions
      WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL`,
    [userId]
  );
  return r.rows[0]?.n ?? 0;
}

async function getBaselineAccuracy(userId: string, beforeDays: number): Promise<number> {
  const r = await pool.query<{ avg: number | null }>(
    `SELECT AVG(accuracy)::float AS avg
       FROM sessions
      WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL
        AND ended_at >= NOW() - ($2 || ' days')::interval
        AND ended_at < NOW()`,
    [userId, String(beforeDays)]
  );
  return r.rows[0]?.avg ?? 0;
}

async function getAccuracySince(userId: string, sinceISO: string): Promise<{
  avg: number;
  sessions: number;
  days: number;
}> {
  const r = await pool.query<{ avg: number | null; n: number; days: number }>(
    `SELECT AVG(accuracy)::float AS avg,
            COUNT(*)::int AS n,
            COUNT(DISTINCT DATE(ended_at))::int AS days
       FROM sessions
      WHERE user_id = $1 AND status='accepted' AND deleted_at IS NULL
        AND ended_at >= $2`,
    [userId, sinceISO]
  );
  const row = r.rows[0];
  return {
    avg: row?.avg ?? 0,
    sessions: row?.n ?? 0,
    days: row?.days ?? 0,
  };
}

/* ═══════════════════════════════════════════════════════════
   SUGGESTIONS
   ═══════════════════════════════════════════════════════════ */
export async function getLabSuggestions(
  userId: string,
  lang: Lang
): Promise<LabSuggestion[]> {
  const cacheKey = `${userId}:${lang}`;
  const cached = suggestionCache.get(cacheKey);
  if (cached && Date.now() - cached.ts < CACHE_TTL_MS) {
    return cached.items;
  }

  const sessionCount = await getUserSessionCount(userId);

  const items: LabSuggestion[] = EXPERIMENT_TEMPLATES.map((t) => {
    const eligible = sessionCount >= t.min_prior_sessions;
    const reason = eligible
      ? undefined
      : lang === 'ar'
        ? `محتاج ${t.min_prior_sessions} جلسات على الأقل (عندك ${sessionCount})`
        : `Need at least ${t.min_prior_sessions} sessions (you have ${sessionCount})`;
    return {
      key: t.key,
      title: templateTitle(t, lang),
      hypothesis: templateHypothesis(t, lang),
      description: templateDescription(t, lang),
      duration_days: t.duration_days,
      eligible,
      ineligible_reason: reason,
    };
  });

  suggestionCache.set(cacheKey, { ts: Date.now(), items });
  return items;
}

export async function startLabExperiment(
  userId: string,
  experimentKey: string,
  lang: Lang
): Promise<LabCurrent> {
  const template = EXPERIMENT_TEMPLATES.find((t) => t.key === experimentKey);
  if (!template) throw new Error('unknown_experiment');

  /* Only one active at a time */
  const active = await pool.query<{ id: string }>(
    `SELECT id FROM lab_experiments
      WHERE user_id = $1 AND status = 'active'
      LIMIT 1`,
    [userId]
  );
  if (active.rowCount && active.rowCount > 0) {
    throw new Error('experiment_already_active');
  }

  /* Check eligibility */
  const sessionCount = await getUserSessionCount(userId);
  if (sessionCount < template.min_prior_sessions) {
    throw new Error('not_enough_sessions');
  }

  /* Baseline: average accuracy over the last 14 days */
  const baseline = await getBaselineAccuracy(userId, 14);

  /* Insert */
  const res = await pool.query<{
    id: string;
    started_at: string;
  }>(
    `INSERT INTO lab_experiments
       (user_id, experiment_key, title, hypothesis, protocol, status, baseline_metric)
     VALUES ($1, $2, $3, $4, $5, 'active', $6)
     RETURNING id, started_at`,
    [
      userId,
      template.key,
      templateTitle(template, lang),
      templateHypothesis(template, lang),
      JSON.stringify(template.protocol),
      baseline,
    ]
  );

  const row = res.rows[0];
  if (!row) throw new Error('lab_insert_failed');

  return {
    experiment_id: row.id,
    experiment_key: template.key,
    title: templateTitle(template, lang),
    hypothesis: templateHypothesis(template, lang),
    description: templateDescription(template, lang),
    duration_days: template.duration_days,

/* ═══════════════════════════════════════════════════════════
   CURRENT
   ═══════════════════════════════════════════════════════════ */
export async function getCurrentLab(
  userId: string,
  lang: Lang
): Promise<LabCurrent | null> {
  const res = await pool.query<{
    id: string;
    experiment_key: string;
    title: string;
    hypothesis: string;
    protocol: Record<string, unknown>;
    status: string;
    baseline_metric: number | null;
    result_metric: number | null;
    result_summary: string | null;
    started_at: string;
    completed_at: string | null;
  }>(
    `SELECT id, experiment_key, title, hypothesis, protocol, status,
            baseline_metric, result_metric, result_summary,
            started_at, completed_at
       FROM lab_experiments
      WHERE user_id = $1 AND status = 'active'
      ORDER BY started_at DESC
      LIMIT 1`,
    [userId]
  );

  const row = res.rows[0];
  if (!row) return null;

  const template = EXPERIMENT_TEMPLATES.find((t) => t.key === row.experiment_key);

  /* Compute current accuracy since started_at */
  const during = await getAccuracySince(userId, row.started_at);
  const startedAt = new Date(row.started_at).getTime();
  const daysElapsed = Math.max(0, Math.floor((Date.now() - startedAt) / 86_400_000));

  const baseline = row.baseline_metric ?? 0;
  const current = during.avg > 0 ? during.avg : null;
  const delta = current !== null ? current - baseline : null;

  const required = template?.duration_days ?? 7;
  const progress = Math.min(100, Math.round((daysElapsed / required) * 100));

  return {
    experiment_id: row.id,
    experiment_key: row.experiment_key,
    title: row.title,
    hypothesis: row.hypothesis,
    description: template ? templateDescription(template, lang) : '',
    duration_days: required,
    status: (row.status as LabCurrent['status']) ?? 'active',
    started_at: row.started_at,
    completed_at: row.completed_at,
    days_elapsed: daysElapsed,
    sessions_completed: during.sessions,
    sessions_required: required,
    progress_pct: progress,
    baseline_accuracy: baseline > 0 ? Math.round(baseline * 10) / 10 : null,
    current_accuracy: current !== null ? Math.round(current * 10) / 10 : null,
    delta: delta !== null ? Math.round(delta * 10) / 10 : null,
    result_summary: row.result_summary,
  };
}

/* ═══════════════════════════════════════════════════════════
   COMPLETE
   ═══════════════════════════════════════════════════════════ */
export async function completeLabExperiment(
  userId: string,
  lang: Lang
): Promise<LabCurrent> {
  const current = await getCurrentLab(userId, lang);
  if (!current) throw new Error('no_experiment');
  if (current.status !== 'active') throw new Error('experiment_not_active');

  /* Require at least 3 sessions to compute result */
  if (current.sessions_completed < 3) {
    throw new Error('not_enough_sessions_to_complete');
  }

  const baseline = current.baseline_accuracy ?? 0;
  const result = current.current_accuracy ?? 0;
  const delta = result - baseline;

  const template = EXPERIMENT_TEMPLATES.find((t) => t.key === current.experiment_key);
  const protocolFollowed =
    current.sessions_completed >= Math.min(3, current.sessions_required);

  /* LLM phrasing */
  const ctx: LabResultContext = {
    title: current.title,
    hypothesis: current.hypothesis,
    baseline_accuracy: baseline > 0 ? Math.round(baseline * 10) / 10 : null,
    result_accuracy: result,
    delta,
    sessions_completed: current.sessions_completed,
    sessions_required: current.sessions_required,
    days_elapsed: current.days_elapsed,
    protocol_followed: protocolFollowed,
  };

  let summary = '';
  const systemPrompt =
    lang === 'ar' ? SYSTEM_PROMPT_LAB_RESULT_AR : SYSTEM_PROMPT_LAB_RESULT_EN;
  const userPrompt = buildLabResultPrompt(ctx, lang);

  try {
    const res = await routeChat(
      [
        { role: 'system', content: systemPrompt },
        { role: 'user', content: userPrompt },
      ],
      { temperature: 0.5, max_tokens: 600, language: lang }
    );
    const check = validateAiText(res.content, { minLen: 50, maxLen: 1200 });
    if (check.ok) {
      summary = check.cleaned;
    }
  } catch (err) {
    console.error('[lab] LLM failed:', (err as Error).message);
  }

  /* Fallback */
  if (!summary) {
    if (lang === 'ar') {
      const verdict =
        delta > 5 ? 'الفرضية اتحققت' : delta < -5 ? 'الفرضية اترفضت' : 'النتيجة غير حاسمة';
      summary = `خلصت التجربة بـ ${current.sessions_completed} جلسات على ${current.days_elapsed} يوم. خط الأساس ${baseline.toFixed(1)}% والنتيجة ${result.toFixed(1)}% (${delta >= 0 ? '+' : ''}${delta.toFixed(1)}%). ${verdict}.`;
    } else {
      const verdict =
        delta > 5 ? 'Hypothesis confirmed' : delta < -5 ? 'Hypothesis refuted' : 'Result inconclusive';
      summary = `Completed the experiment with ${current.sessions_completed} sessions over ${current.days_elapsed} days. Baseline ${baseline.toFixed(1)}% vs result ${result.toFixed(1)}% (${delta >= 0 ? '+' : ''}${delta.toFixed(1)}%). ${verdict}.`;
    }
  }

  /* Update DB */
  await pool.query(
    `UPDATE lab_experiments
        SET status = 'completed',
            result_metric = $1,
            result_summary = $2,
            completed_at = NOW()
      WHERE id = $3`,
    [result, summary, current.experiment_id]
  );

  const fresh = await getCurrentLab(userId, lang);
  if (!fresh) throw new Error('lab_refresh_failed');
  return fresh;
}

/* ═══════════════════════════════════════════════════════════
   AI Router — Multi-provider failover with circuit breaker
   Order: Groq -> OpenRouter -> Gemini
   Records health per provider; skips unhealthy ones.
   Supports language-aware model selection per provider.
   ═══════════════════════════════════════════════════════════ */

import { groqProvider } from './providers/groq-provider.js';
import { openrouterProvider } from './providers/openrouter-provider.js';
import { geminiProvider } from './providers/gemini-provider.js';
import { isHealthy, recordSuccess, recordFailure, getHealthReport } from './health.js';
import type { AIProvider, AIMessage, AIOptions, AIResult, Lang } from './providers/types.js';
import { env } from '../../config/env.js';

/* Provider priority order */
const PROVIDERS: AIProvider[] = [groqProvider, openrouterProvider, geminiProvider];

export interface RouteOptions extends AIOptions {
  /** Language hint — used to pick the best model per provider */
  language?: Lang;
  /** Providers already attempted (used internally by routeChat) */
  exclude?: string[];
}

export interface RouteResult extends AIResult {
  fallback_chain: string[];
}

/* ─── Model picker per provider ─── */
function pickModel(providerName: string, language: Lang | undefined, explicitModel?: string): string | undefined {
  if (explicitModel) return explicitModel;

  if (providerName === 'groq') {
    if (language === 'ar') return env.GROQ_MODEL_AR;
    return env.GROQ_MODEL_EN;
  }

  if (providerName === 'openrouter') {
    return env.OPENROUTER_MODEL; // 'meta-llama/llama-3.3-70b-instruct:free'
  }

  if (providerName === 'gemini') {
    return env.GEMINI_MODEL; // 'gemini-2.0-flash'
  }

  return undefined;
}

/**
 * Route a chat request through available providers.
 * Tries in priority order, skipping unhealthy ones.
 * Throws if all providers fail.
 */
export async function routeChat(
  messages: AIMessage[],
  opts: RouteOptions = {}
): Promise<RouteResult> {
  const exclude = new Set(opts.exclude ?? []);
  const attempted: string[] = [];
  const errors: string[] = [];

  for (const provider of PROVIDERS) {
    if (exclude.has(provider.name)) continue;
    if (!provider.isEnabled()) continue;
    if (!isHealthy(provider.name)) {
      errors.push(`${provider.name}: unhealthy (cooldown)`);
      continue;
    }

    const model = pickModel(provider.name, opts.language, opts.model);
    attempted.push(provider.name);

    console.error(
      '[router] trying provider=' + provider.name,
      '| model=' + (model || '(default)'),
      '| lang=' + (opts.language || 'n/a')
    );

    try {
      const result = await provider.chat(messages, {
        temperature: opts.temperature,
        max_tokens: opts.max_tokens,
        json_mode: opts.json_mode,
        model,
      });

      if (!result.content || result.content.trim().length === 0) {
        throw new Error('empty response');
      }

      console.error('[router] SUCCESS via', provider.name, '| len:', result.content.length);
      recordSuccess(provider.name);
      return {
        ...result,
        fallback_chain: attempted,
      };
    } catch (err) {
      const msg = (err as Error).message;
      console.error('[router] FAILED via', provider.name, ':', msg.slice(0, 200));
      errors.push(`${provider.name}: ${msg}`);
      recordFailure(provider.name, msg);
    }
  }

  throw new Error(`All AI providers failed:\n${errors.join('\n')}`);
}

/**
 * Route a JSON-mode request. Parses and returns object.
 */
export async function routeJson<T>(
  messages: AIMessage[],
  opts: RouteOptions = {}
): Promise<{ data: T; meta: RouteResult }> {
  const result = await routeChat(messages, { ...opts, json_mode: true });

  let cleaned = result.content.trim();
  /* Strip markdown fences if any */
  const fenceMatch = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/i);
  if (fenceMatch) cleaned = fenceMatch[1].trim();
  /* Fallback: find first {...last } */
  if (!cleaned.startsWith('{')) {
    const first = cleaned.indexOf('{');
    const last = cleaned.lastIndexOf('}');
    if (first !== -1 && last > first) {
      cleaned = cleaned.slice(first, last + 1);
    }
  }

  let parsed: T;
  try {
    parsed = JSON.parse(cleaned) as T;
  } catch {
    throw new Error(`Router: invalid JSON from ${result.provider}: ${result.content.slice(0, 200)}`);
  }

  return { data: parsed, meta: result };
}

export function getProviderStatus() {
  return {
    providers: PROVIDERS.map((p) => ({
      name: p.name,
      enabled: p.isEnabled(),
    })),
    health: getHealthReport(),
  };
}
