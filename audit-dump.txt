/*
 * Neon (serverless) friendly pool.
 * - max: 10 balances Neon free tier (100 CU-hours) with burst tolerance
 * - statement_timeout: kill slow queries before they burn compute
 * - idle_in_transaction_session_timeout: prevent abandoned transactions
 * - keepAlive: reuse TCP, avoid reconnecting on every request
 */

  connectionString: env.DATABASE_URL,
  max: 10,
  idleTimeoutMillis: 20_000,
  connectionTimeoutMillis: 10_000,
  keepAlive: true,
  application_name: 'flash-backend',
  statement_timeout: 15_000,
  idle_in_transaction_session_timeout: 30_000,
  ssl: env.NODE_ENV === 'production' ? { rejectUnauthorized: false } : false,
});

pool.on('error', (err) => {
  console.error('[DB] Unexpected pool error:', err.message);
});

export async function ping(): Promise<boolean> {
  try {
    const result = await pool.query('SELECT 1 AS ok');
    return result.rows[0]?.ok === 1;
  } catch {
    return false;
  }
}

export async function verifyOTP(
  email: string,
  code: string,
  meta: { device_id?: string; user_agent?: string; ip?: string }
): Promise<{ sessionToken: string; playerId: string; expiresAt: string }> {

  const r = await pool.query<{
    id: string;
    code_hash: string;
    attempts: number;
    expires_at: string;
    consumed_at: string | null;
  }>(

       FROM auth_otp_codes

  if (new Date(row.expires_at).getTime() < Date.now()) throw new Error('otp_expired');
  if (row.attempts >= OTP_MAX_ATTEMPTS) throw new Error('too_many_attempts');

  /* Constant-time compare */
  const a = Buffer.from(row.code_hash, 'hex');
  const b = Buffer.from(codeHash, 'hex');
  const ok = a.length === b.length && timingSafeEqual(a, b);

  if (!ok) {
    await pool.query(
      `UPDATE auth_otp_codes SET attempts = attempts + 1 WHERE id = $1`,
      [row.id]
    );
    throw new Error('otp_invalid');
  }

  await pool.query(
    `UPDATE auth_otp_codes SET consumed_at = now() WHERE id = $1`,
    [row.id]
  );

  /* Resolve player + session */
  const playerId = await getOrCreatePlayerByEmail(norm);
  await linkIdentity(playerId, 'email_otp', norm);

  const token = generateSessionToken();
  const tokenHash = hashHex(token);
  const idleExp = new Date(Date.now() + SESSION_IDLE_MS);
  const absExp  = new Date(Date.now() + SESSION_ABS_MS);

  await pool.query(

function median(arr: number[]): number | null {
  if (arr.length === 0) return null;
  const sorted = [...arr].sort((a, b) => a - b);
  const mid = Math.floor(sorted.length / 2);
  return sorted.length % 2 === 0
    ? (sorted[mid - 1]! + sorted[mid]!) / 2
    : sorted[mid]!;
}

  const correctTrials = valid.filter((t) => t.correct === true);

    .map((t) => t.rt_ms)

  const accuracy = n_trials > 0 ? (n_correct / n_trials) * 100 : 0;
  const median_rt_ms = median(rts);
  const total_rt_seconds = rts.reduce((a, b) => a + b, 0) / 1000;
  const rcs = total_rt_seconds > 0 ? n_correct / total_rt_seconds : null;

  let primary_score = 0;
  const metric = game?.primary_metric ?? 'accuracy';
  switch (metric) {
    case 'accuracy':
    case 'hard_emotion_accuracy':
    case 'novelty_score':

      primary_score = accuracy;
      break;
    case 'median_rt_ms':
      primary_score = median_rt_ms ? Math.round(1000 / (median_rt_ms / 1000)) : 0;
      break;
    case 'max_span':
    case 'max_n_level':
      primary_score = accuracy;
      break;
    default:
      primary_score = accuracy;
  }

  if (game?.max_score && primary_score > game.max_score) {
    primary_score = game.max_score;
  }

    n_trials,
    n_correct,
    accuracy: Math.round(accuracy * 100) / 100,
    median_rt_ms: median_rt_ms !== null ? Math.round(median_rt_ms) : null,
    rcs: rcs !== null ? Math.round(rcs * 10000) / 10000 : null,
    primary_score: Math.round(primary_score * 100) / 100,
  };

/* ============================================================
   FLASH Backend - Idempotency Middleware
   Prevents duplicate mutations when client retries.
   Uses in-memory LRU with TTL. Safe, no DB writes.
   ============================================================ */
import type { FastifyRequest, FastifyReply } from 'fastify';
import { createHash } from 'node:crypto';

interface CacheEntry {
  status: number;
  body: string;
  contentType: string;
  createdAt: number;
}

const CACHE = new Map<string, CacheEntry>();
const TTL_MS = 5 * 60 * 1000;
const MAX_ENTRIES = 500;

function hashKey(key: string): string {
  return createHash('sha256').update(key).digest('hex').slice(0, 32);
}

function gc() {
  const now = Date.now();
  for (const [k, v] of CACHE) {
    if (now - v.createdAt > TTL_MS) CACHE.delete(k);
  }
  if (CACHE.size > MAX_ENTRIES) {
    const sorted = [...CACHE.entries()].sort((a, b) => a[1].createdAt - b[1].createdAt);
    const drop = sorted.slice(0, CACHE.size - MAX_ENTRIES);
    for (const [k] of drop) CACHE.delete(k);
  }
}

export async function idempotencyHook(req: FastifyRequest, reply: FastifyReply): Promise<void> {
  const raw = req.headers['idempotency-key'];
  if (!raw || typeof raw !== 'string') return;
  if (req.method === 'GET' || req.method === 'HEAD' || req.method === 'OPTIONS') return;

  const hashed = hashKey(raw);
  gc();

  const cached = CACHE.get(hashed);
  if (cached) {
    reply
      .code(cached.status)
      .header('content-type', cached.contentType)
      .header('x-idempotent-replay', '1')
      .send(cached.body);
    return;
  }

  const originalSend = reply.send.bind(reply);
  reply.send = function (payload: unknown) {
    try {
      if (reply.statusCode < 500 && reply.statusCode !== 409) {
        const body = typeof payload === 'string' ? payload : JSON.stringify(payload);
        CACHE.set(hashed, {
          status: reply.statusCode,
          body,
          contentType: (reply.getHeader('content-type')?.toString()) || 'application/json',
          createdAt: Date.now()
        });
      }
    } catch (_) { /* never break response on cache error */ }
    return originalSend(payload);
  } as typeof reply.send;
}

/**
 * Weighted staircase:
 *   promote only after PROMOTE_NEEDED consecutive good rounds
 *   demote immediately if accuracy < DEMOTE_THRESHOLD
 *   frustration cap: 3+ fails → recovery round (stay, but mark for recovery)
 */
export function decideNextLevel(
  current: Ability,
  accuracy: number,
  _medianRT: number | null | undefined
): Decision {
  const from = current.current_level;

  /* Frustration cap */
  if (current.consecutive_failures >= FRUSTRATION_CAP) {
    const to = Math.max(MIN_LEVEL, from - 1);
    return {
      from_level: from,
      to_level: to,
      reason_code: 'HOLD_FRUSTRATION',
      explanation: `Staying at level ${to} to rebuild momentum — you've had ${current.consecutive_failures} tough rounds in a row.`,
      predicted_success: 0.85,
    };
  }

  /* Struggling → demote */
  if (accuracy < DEMOTE_THRESHOLD) {
    const to = Math.max(MIN_LEVEL, from - 1);
    return {
      from_level: from,
      to_level: to,
      reason_code: 'DEMOTE_STRUGGLE',
      explanation: `Moving to level ${to} — the current level felt too hard (${Math.round(accuracy)}% accuracy).`,
      predicted_success: 0.75,
    };
  }

  /* Doing well → check streak for promotion */
  if (accuracy >= PROMOTE_THRESHOLD) {
    const newStreak = current.consecutive_successes + 1;
    if (newStreak >= PROMOTE_NEEDED && from < MAX_LEVEL) {
      const to = Math.min(MAX_LEVEL, from + 1);
      return {
        from_level: from,
        to_level: to,
        reason_code: 'PROMOTE_STABLE',
        explanation: `Moving to level ${to} — you've hit ${PROMOTE_THRESHOLD}%+ accuracy for ${newStreak} rounds in a row.`,
        predicted_success: 0.78,
      };
    }
    return {
      from_level: from,
      to_level: from,
      reason_code: 'HOLD_IN_BAND',
      explanation: `Staying at level ${from} — great accuracy (${Math.round(accuracy)}%), but let's confirm it's stable.`,
      predicted_success: 0.80,
    };
  }

  /* In target band → hold */
  return {
    from_level: from,
    to_level: from,
    reason_code: 'HOLD_IN_BAND',
    explanation: `Staying at level ${from} — you're in your target zone (${Math.round(accuracy)}% accuracy).`,
    predicted_success: 0.80,
  };
}

/**

  const alpha = Math.max(0.15, 1 / Math.max(3, current.evidence_count));

/**
 * Route a chat request through available providers.
 * Tries in priority order, skipping unhealthy ones.
 * Throws if all providers fail.
 */
export async function routeChat(
  messages: AIMessage[],
  opts: RouteOptions = {}
): Promise<RouteResult> {
  const exclude = new Set(opts.exclude ?? []);
  const attempted: string[] = [];
  const errors: string[] = [];

  for (const provider of PROVIDERS) {
    if (exclude.has(provider.name)) continue;
    if (!provider.isEnabled()) continue;
    if (!isHealthy(provider.name)) {
      errors.push(`${provider.name}: unhealthy (cooldown)`);
      continue;
    }

    const model = pickModel(provider.name, opts.language, opts.model);
    attempted.push(provider.name);

    console.error(
      '[router] trying provider=' + provider.name,
      '| model=' + (model || '(default)'),
      '| lang=' + (opts.language || 'n/a')
    );

    try {
      const result = await provider.chat(messages, {
        temperature: opts.temperature,
        max_tokens: opts.max_tokens,
        json_mode: opts.json_mode,
        model,
      });

      if (!result.content || result.content.trim().length === 0) {
        throw new Error('empty response');
      }

      console.error('[router] SUCCESS via', provider.name, '| len:', result.content.length);
      recordSuccess(provider.name);
      return {
        ...result,
        fallback_chain: attempted,
      };
    } catch (err) {
      const msg = (err as Error).message;
      console.error('[router] FAILED via', provider.name, ':', msg.slice(0, 200));
      errors.push(`${provider.name}: ${msg}`);
      recordFailure(provider.name, msg);
    }
  }

  throw new Error(`All AI providers failed:\n${errors.join('\n')}`);
}

/**
 * Route a JSON-mode request. Parses and returns object.
 */
export async function routeJson<T>(
  messages: AIMessage[],
  opts: RouteOptions = {}
): Promise<{ data: T; meta: RouteResult }> {
  const result = await routeChat(messages, { ...opts, json_mode: true });

  let cleaned = result.content.trim();
  /* Strip markdown fences if any */
  const fenceMatch = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/i);
  if (fenceMatch && fenceMatch[1]) cleaned = fenceMatch[1].trim();
  /* Fallback: find first {...last } */
  if (!cleaned.startsWith('{')) {
    const first = cleaned.indexOf('{');
    const last = cleaned.lastIndexOf('}');
    if (first !== -1 && last > first) {
      cleaned = cleaned.slice(first, last + 1);
    }
  }

  let parsed: T;
  try {
    parsed = JSON.parse(cleaned) as T;
  } catch {
    throw new Error(`Router: invalid JSON from ${result.provider}: ${result.content.slice(0, 200)}`);
  }

  return { data: parsed, meta: result };
}

    this.name = 'AppError';
    this.code = opts.code;
    this.category = opts.category;
    this.status = opts.status;
    this.retryable = opts.retryable ?? false;
    this.userMessage = opts.userMessage ?? defaultUserMessage(opts.category);
    this.requestId = opts.requestId;
    this.context = opts.context ?? {};
    Error.captureStackTrace?.(this, AppError);
  }
}

/* ============================================================
   POSTGRES ERROR MAPPING
   ============================================================ */

interface PgError {
  code?: string;
  constraint?: string;
  detail?: string;
  table?: string;
}

export function mapPgError(err: unknown): AppError {
  if (err instanceof AppError) return err;

  const pgErr = err as PgError;
  const code = pgErr?.code;

  switch (code) {
    case '23505': /* unique_violation */
      return new AppError({
        code: 'DUPLICATE_RECORD',
        category: 'SAVE_SYNC',
        status: 409,
        message: 'Duplicate record',
        context: { constraint: pgErr.constraint }
      });
    case '23503': /* foreign_key_violation */
      return new AppError({
        code: 'FK_VIOLATION',
        category: 'VALIDATION',
        status: 400,
        message: 'Reference does not exist',
        context: { constraint: pgErr.constraint }
      });
    case '23502': /* not_null_violation */
      return new AppError({
        code: 'NOT_NULL_VIOLATION',
        category: 'VALIDATION',
        status: 422,
        message: 'Required field is missing'
      });
    case '23514': /* check_violation */
      return new AppError({
        code: 'CHECK_VIOLATION',
        category: 'VALIDATION',
        status: 422,
        message: 'Value violates constraint'
      });
    case '40001': /* serialization_failure */
    case '40P01': /* deadlock_detected */
      return new AppError({
        code: 'TRANSIENT_CONFLICT',
        category: 'SAVE_SYNC',
        status: 503,
        message: 'Transient conflict',
        retryable: true
      });
    case '57014': /* query_canceled */
      return new AppError({
        code: 'QUERY_TIMEOUT',
        category: 'SERVER',
        status: 504,
        message: 'Query timed out',
        retryable: true
      });
    case '53300': /* too_many_connections */
      return new AppError({
        code: 'POOL_EXHAUSTED',
        category: 'SERVER',
        status: 503,
        message: 'Database pool exhausted',
        retryable: true
      });
    default:
      return new AppError({
        code: 'DB_ERROR',
        category: 'SERVER',
        status: 500,
        message: 'Database error',
        context: { pgCode: code || 'unknown' }
      });
  }
}

import type { FastifyInstance, FastifyError, FastifyRequest, FastifyReply } from 'fastify';
import { AppError, mapPgError, toResponse } from './errors.js';
import { randomUUID } from 'node:crypto';

export async function registerErrorHandler(app: FastifyInstance): Promise<void> {
  /* Generate request ID if not present */
  app.addHook('onRequest', async (req) => {
    if (!req.headers['x-request-id']) {
      (req as FastifyRequest & { id: string }).id = randomUUID();
    }
  });

  app.setErrorHandler((err: FastifyError, req: FastifyRequest, reply: FastifyReply) => {
    const requestId =
      (req.headers['x-request-id'] as string) ||
      (req as FastifyRequest & { id: string }).id ||
      undefined;

    /* Map to AppError */
    let appErr: AppError;

    if (err instanceof AppError) {
      appErr = err;
    } else if (err.validation) {
      appErr = new AppError({
        code: 'VALIDATION_ERROR',
        category: 'VALIDATION',
        status: 400,
        message: 'Request validation failed',
        context: { details: err.validation }
      });
    } else if (err.statusCode === 429) {
      appErr = new AppError({
        code: 'RATE_LIMIT',
        category: 'RATE_LIMIT',
        status: 429,
        message: 'Too many requests',
        retryable: true
      });
    } else if (err.statusCode === 413) {
      appErr = new AppError({
        code: 'PAYLOAD_TOO_LARGE',
        category: 'VALIDATION',
        status: 413,
        message: 'Request payload too large'
      });
    } else if ((err as { code?: string }).code && /^[0-9A-Z]{5}$/.test((err as { code?: string }).code || '')) {
      /* Postgres error shape */
      appErr = mapPgError(err);
    } else if (err.statusCode && err.statusCode >= 400 && err.statusCode < 500) {
      appErr = new AppError({
        code: 'CLIENT_ERROR',
        category: 'VALIDATION',
        status: err.statusCode,
        message: err.message || 'Client error'
      });
    } else {
      appErr = new AppError({
        code: 'SERVER_ERROR',
        category: 'SERVER',
        status: 500,
        message: 'Internal server error',
        retryable: true
      });
    }

    /* Log safely (never leak secrets) */
    const logLevel = appErr.status >= 500 ? 'error' : 'warn';
    req.log[logLevel]({
      err: {
        code: appErr.code,
        category: appErr.category,
        status: appErr.status,
        message: appErr.message
      },
      requestId,
      context: appErr.context,
      path: req.url,
      method: req.method
    }, 'Request error');

    /* Send response */
    const body = toResponse(appErr, requestId);

    reply
      .code(appErr.status)
      .header('x-request-id', requestId || '')
      .header('content-type', 'application/json; charset=utf-8')
      .send(body);
  });

  /* 404 handler */
  app.setNotFoundHandler((req, reply) => {
    const requestId =
      (req.headers['x-request-id'] as string) ||
      (req as FastifyRequest & { id: string }).id ||
      undefined;

    reply
      .code(404)
      .header('x-request-id', requestId || '')
      .send({
        error: 'NOT_FOUND',
        code: 'NOT_FOUND',
        category: 'GAME_STATE',
        message: 'Resource not found',
        retryable: false,
        requestId
      });
  });
}

   FLASH - API Client (v2 - Error-System integrated)
   Uses FLASH_SAFE_FETCH for all network calls
   Auto auth header, throws classified errors, never leaks
   ============================================================ */
(function () {
  'use strict';

  const ERR = window.FLASH_ERRORS;
  const SF  = window.FLASH_SAFE_FETCH;

  if (!ERR || !SF) {
    console.error('[api] requires errors.js + safe-fetch.js');
    return;
  }

  const API_BASE =
    (window.FLASH_CONFIG && window.FLASH_CONFIG.API_BASE) ||
    'http://localhost:3000';

  /* ============================================================
     AUTH
     ============================================================ */
  function getToken() {
    try { return localStorage.getItem('flash.session'); } catch (_) { return null; }
  }

  function authHeader() {
    const t = getToken();
    return t ? { Authorization: 'Bearer ' + t } : {};
  }

  /* ============================================================
     CORE REQUEST
     ============================================================ */
  async function request(path, opts) {
    opts = opts || {};
    const url = /^https?:\/\//.test(path) ? path : API_BASE + path;

    const headers = Object.assign(
      { 'Content-Type': 'application/json' },
      authHeader(),
      opts.headers || {}
    );

    const controller = new AbortController();
    if (opts.signal) {
      if (opts.signal.aborted) controller.abort();
      else opts.signal.addEventListener('abort', () => controller.abort(), { once: true });
    }

    try {
      const res = await SF(url, Object.assign({}, opts, {
        headers,
        signal: controller.signal,
        idempotencyKey: opts.idempotencyKey
      }));

      return await parseBody(res);
    } catch (err) {
      /* Already classified by safe-fetch */
      throw enrichError(err, opts);
    }
  }

  async function parseBody(res) {
    if (res.status === 204) return null;
    const ct = res.headers.get('content-type') || '';
    if (ct.indexOf('application/json') >= 0) {
      try { return await res.json(); }
      catch (e) {
        ERR.log('INTEGRATION', {
          message: 'Invalid JSON in response',
          operation: res.url,
          status: res.status
        });
        return null;
      }
    }
    return res.text();
  }

  function enrichError(err, opts) {
    const cls = ERR.classify(err);
    /* Attach category for caller convenience */
    err.category = cls.category;
    err.retryable = cls.retryable;
    err.aborted = cls.aborted;

    /* Auto-show toast for errors that aren't handled by caller */
    if (!opts.silent) {
      const UI = window.FLASH_UI;
      if (UI && !cls.aborted) {
        UI.showToast({
          kind: 'error',
          category: cls.category,
          title: opts.errorTitle || null
        });
      }
    }
    return err;
  }

     ============================================================ */
  const API = {

/* ============================================================
   FLASH - Safe Fetch Wrapper
   Timeout + retry with backoff + idempotency + classification
   ============================================================ */
(function () {
  'use strict';

  const ERR = window.FLASH_ERRORS;
  if (!ERR) { console.error('[safe-fetch] errors.js required'); return; }

  const CFG = {
    timeoutMs: 30000,
    retries: 2,
    baseDelayMs: 600,
    maxDelayMs: 8000,
    retryableStatuses: [408, 429, 500, 502, 503, 504]
  };

  function sleep(ms) { return new Promise((r) => setTimeout(r, ms)); }

  function backoff(attempt) {
    const base = Math.min(CFG.baseDelayMs * Math.pow(2, attempt), CFG.maxDelayMs);
    return base + Math.random() * 0.3 * base;
  }

  function isIdempotentMethod(method) {
    const m = (method || 'GET').toUpperCase();
    return m === 'GET' || m === 'HEAD' || m === 'OPTIONS';
  }

  function safeFetch(url, options) {
    const opts = Object.assign({}, options || {});
    const method = (opts.method || 'GET').toUpperCase();
    const idempotent = isIdempotentMethod(method);
    const hasIdempKey = !!opts.idempotencyKey;
    const maxRetries = (idempotent || hasIdempKey) ? CFG.retries : 0;

    const controller = new AbortController();
    const timeoutId = setTimeout(() => {
      try { controller.abort(); } catch (_) {}
    }, opts.timeoutMs || CFG.timeoutMs);

    if (opts.signal) {
      if (opts.signal.aborted) controller.abort();
      else opts.signal.addEventListener('abort', () => controller.abort(), { once: true });
    }

    const headers = Object.assign({}, opts.headers || {});
    if (hasIdempKey) headers['Idempotency-Key'] = opts.idempotencyKey;
    if (!headers['Content-Type'] && opts.body && typeof opts.body === 'string') {
      headers['Content-Type'] = 'application/json';
    }

    let lastErr = null;

    async function attempt(attemptNum) {
      const fetchOpts = Object.assign({}, opts, { headers, signal: controller.signal });
      delete fetchOpts.idempotencyKey;
      delete fetchOpts.timeoutMs;

      try {
        const res = await fetch(url, fetchOpts);

        if (res.ok) {
          clearTimeout(timeoutId);
          return res;
        }

        const shouldRetry = CFG.retryableStatuses.indexOf(res.status) >= 0;

        if (shouldRetry && attemptNum < maxRetries) {
          await sleep(backoff(attemptNum));
          return attempt(attemptNum + 1);
        }

        const error = new Error('HTTP ' + res.status);
        error.status = res.status;
        error.response = res;

        try {
          const ct = res.headers.get('content-type') || '';
          if (ct.indexOf('application/json') >= 0) {
            error.body = await res.clone().json();
          }
        } catch (_) {}

        const cls = ERR.classify(error);
        ERR.log(cls.category, {
          status: res.status,
          operation: method + ' ' + url,
          requestId: res.headers.get('x-request-id') || null,
          message: 'HTTP ' + res.status
        });

        throw error;
      } catch (err) {
        clearTimeout(timeoutId);

        if (err.name === 'AbortError') {
          const abortErr = new Error('Request aborted or timed out');
          abortErr.name = 'AbortError';
          abortErr.status = 0;
          throw abortErr;
        }

        lastErr = err;

        const status = err.status || 0;
        const retryable = !status || status >= 500 || status === 408 || status === 429;

        if (retryable && attemptNum < maxRetries) {
          await sleep(backoff(attemptNum));
          return attempt(attemptNum + 1);
        }

        throw err;
      }
    }

    return attempt(0);
  }

  /* Helpers */
  safeFetch.json = async function (url, options) {
    const res = await safeFetch(url, options);
    const ct = res.headers.get('content-type') || '';
    if (ct.indexOf('application/json') >= 0) return res.json();
    return null;
  };

  safeFetch.text = async function (url, options) {
    const res = await safeFetch(url, options);
    return res.text();
  };

  window.FLASH_SAFE_FETCH = safeFetch;
  window.safeFetch = safeFetch;
})();

  function startClassic() {
    state.mode = 'classic';
    state.level = 1;
    state.lives = CONFIG.maxLives;
    state.streak = 0;
    state.maxStreak = 0;
    state.maxSpanReached = CONFIG.startSpan;
    state.results = [];
    state.totalScore = 0;
    state.rtList = [];
    state.invalidRTs = 0;

  }

  function startLevel() {
    clearAllTimeouts();
    state.levelCorrect = 0;
    state.levelTotal = 0;
    state.trialIdx = 0;
    state.inputLocked = false;

    const span = spanForLevel(state.level);
    state.maxSpanReached = Math.max(state.maxSpanReached, span);

    // Generate sequences for this level
    state.sequences = [];
    for (let i = 0; i < CONFIG.trialsPerLevel; i++) {
      state.sequences.push(generateSequence(span));
    }

    elLevel.textContent = String(state.level);
    elSpan.textContent = String(span);

    setPhase('countdown');
    showCountdown(3, 'Level ' + state.level, span + ' digits · ' + CONFIG.trialsPerLevel + ' trials', nextTrial);
  }

  function nextTrial() {
    if (state.lives <= 0) { completeGame(); return; }
    if (state.trialIdx >= CONFIG.trialsPerLevel) { endLevel(); return; }

    state.trialIdx++;
    state.currentSeq = state.sequences[state.trialIdx - 1];
    state.userInput = [];
    state.inputLocked = false;

    // Show first digit
    displayDigit(0);
  }

  function displayDigit(idx) {
    if (state.phase === 'gameover') return;
    if (idx >= state.currentSeq.length) {
      // All digits shown — go to input phase
      push(showInputPhase, CONFIG.digitGapDuration);
      return;
    }

    setPhase('stimulus');
    render(renderDigit(state.currentSeq[idx], idx, state.currentSeq.length));
    if (S && S.digitFlash) S.digitFlash(state.currentSeq[idx]);

    // Show digit for digitShowDuration, then gap, then next
    push(() => displayDigit(idx + 1), CONFIG.digitShowDuration + CONFIG.digitGapDuration);
  }

  function showInputPhase() {
    if (state.phase === 'gameover') return;
    setPhase('trial');
    state.userInput = [];
    state.inputStartTime = performance.now();
    state.inputLocked = false;
    render(renderInput());
    wireKeypad();
  }

  /* ═══ KEY HANDLING ═══ */
  function handleKey(key) {
    if (state.mode !== 'classic') return;
    if (state.phase !== 'trial') return;
    if (state.inputLocked) return;

    // Backspace
    if (key === '⌫') {
      if (state.userInput.length > 0) {
        state.userInput.pop();
        if (S && S.keyDelete) S.keyDelete();
        render(renderInput());
        wireKeypad();
      }
      return;
    }

    // Manual submit
    if (key === '✓') {
      if (state.userInput.length === state.currentSeq.length) {
        submitResponse();
      }
      return;
    }

    // Digit input
    if (/^[0-9]$/.test(key)) {
      if (state.userInput.length >= state.currentSeq.length) return;
      state.userInput.push(parseInt(key, 10));
      if (S && S.keyPress) S.keyPress();
      render(renderInput());
      wireKeypad();

      // Auto-submit when full
      if (state.userInput.length === state.currentSeq.length) {
        state.inputLocked = true;
        push(submitResponse, 320);
      }
    }
  }

  function submitResponse() {
    if (state.phase !== 'trial') return;
    if (state.inputLocked && state.userInput.length !== state.currentSeq.length) return;
    state.inputLocked = true;

    const rt = performance.now() - state.inputStartTime;
    const isValid = rt >= CONFIG.rtMin && rt <= CONFIG.rtMax;
    if (!isValid) state.invalidRTs++;

    const isCorrect =
      state.userInput.length === state.currentSeq.length &&
      state.userInput.every((d, i) => d === state.currentSeq[i]);

    state.levelTotal++;

    if (isCorrect) {
      state.levelCorrect++;
      state.streak++;
      if (state.streak > state.maxStreak) state.maxStreak = state.streak;

      if (isValid) state.rtList.push(rt);

      const spanBonus = state.currentSeq.length * CONFIG.spanBonus;
      const streakBonus = (state.streak - 1) * CONFIG.streakBonus;
      state.totalScore += CONFIG.correctBase + spanBonus + streakBonus;

      if (S && S.sequenceComplete) S.sequenceComplete();
      showFeedbackFlash(true);
      haptic('correct');
      setPhase('correct');
      push(nextTrial, 600);
    } else {
      state.streak = 0;

      showFeedbackFlash(false);
      haptic('wrong');
      setPhase('wrong');
      push(nextTrial, 1000);
    }
  }

  /* ═══ END LEVEL ═══ */
  function endLevel() {
    clearAllTimeouts();

    const accuracy = state.levelTotal > 0 ? state.levelCorrect / state.levelTotal : 0;
    const passed = accuracy >= CONFIG.passRatio;

    state.results.push({
      level: state.level,
      correct: state.levelCorrect,
      total: state.levelTotal,
      span: spanForLevel(state.level),
      passed
    });

    if (passed) {
      if (S) S.levelUp();
      setPhase('correct');
      render('' +
        renderProgressDots() +
        '<div class="feedback-icon good">✓</div>' +
        '<h2 class="phase-headline grd">Level ' + state.level + ' Passed</h2>' +
        '<p class="phase-sub">' +
          state.levelCorrect + '/' + state.levelTotal + ' sequences recalled · ' +
          'Span ' + spanForLevel(state.level) + ' digits' +
        '</p>' +
        '<p class="phase-hint">' +
          (state.level < CONFIG.maxLevel ? 'Level ' + (state.level + 1) + ' starting…' : 'You reached the top!') +
        '</p>'
      );

      push(() => {
        if (state.level < CONFIG.maxLevel) {
          state.level++;
          elLevel.textContent = String(state.level);
          startLevel();
        } else {
          completeGame();
        }
      }, 1500);
    } else {
      state.lives--;
      elLives.textContent = String(state.lives);
      if (S) S.gameOver();
      setPhase('wrong');
      render('' +
        renderProgressDots() +
        '<div class="feedback-icon bad">✕</div>' +
        '<h2 class="phase-headline" style="color:#FF0077">Level Failed</h2>' +
        '<p class="phase-sub">' +
          state.levelCorrect + '/' + state.levelTotal + ' sequences recalled — need ' +
          Math.ceil(CONFIG.trialsPerLevel * CONFIG.passRatio) + ' correct.' +
          '<br><strong>' + state.lives + '</strong> ' +
          (state.lives === 1 ? 'life' : 'lives') + ' remaining.' +
        '</p>'
      );

      push(() => {
        if (state.lives > 0) startLevel();
        else completeGame();
      }, 1900);
    }
  }

  function completeGame() {

  function bindGlobalKeys() {
    document.addEventListener('keydown', (e) => {
      const tag = (e.target.tagName || '').toLowerCase();
      if (tag === 'input' || tag === 'select' || tag === 'textarea') return;

      const k = e.key;

      // Idle: SPACE starts classic
      if (state.phase === 'idle' && state.mode === 'idle') {
        if (e.code === 'Space') {
          e.preventDefault();
          startClassic();
        }
        return;
      }

      // Gameover: SPACE restarts
      if (state.phase === 'gameover') {
        if (e.code === 'Space') {
          e.preventDefault();
          resetGame();
        }
        return;
      }

      // Only classic mode handles digits here
      if (state.mode !== 'classic') return;
      if (state.phase !== 'trial') return;
      if (state.inputLocked) return;

      if (/^[0-9]$/.test(k)) {
        e.preventDefault();
        handleKey(k);
      } else if (k === 'Backspace') {
        e.preventDefault();
        handleKey('⌫');
      } else if (k === 'Enter') {
        e.preventDefault();
        handleKey('✓');
      }
    });

(function () {
  'use strict';

  let GAME_ID = null;
  let STAGES = [];
  let STORAGE_KEY = null;
  let onStageComplete = null;

  function safeGet(k, d) { try { const r = localStorage.getItem(k); return r ? JSON.parse(r) : d; } catch { return d; } }
  function safeSet(k, v) { try { localStorage.setItem(k, JSON.stringify(v)); } catch {} }

  /* ═══ INIT ═══ */
  function init(gameId, stages, opts) {
    GAME_ID = gameId;
    STAGES = stages || [];
    STORAGE_KEY = `flash:stages:${gameId}`;
    onStageComplete = (opts && opts.onStageComplete) || null;
    return getProgress();
  }

  /* ═══ PROGRESS ═══ */
  function getProgress() {
    const saved = safeGet(STORAGE_KEY, { completed: [] });
    const completed = saved.completed || [];
    const currentIdx = STAGES.findIndex(s => !completed.includes(s.id));
    return {
      completed: completed,
      total: STAGES.length,
      currentIndex: currentIdx === -1 ? STAGES.length - 1 : currentIdx,
      currentStage: currentIdx === -1 ? null : STAGES[currentIdx],
      allDone: completed.length >= STAGES.length,
      completedCount: completed.length
    };
  }

  /* ═══ CHECK SESSION ═══ */
  function checkSession(payload) {
    const saved = safeGet(STORAGE_KEY, { completed: [] });
    const completed = saved.completed || [];
    const newlyCompleted = [];

    // Only check the current (first not-yet-completed) stage
    // This keeps progression linear and prevents skipping
    const nextStage = STAGES.find(s => !completed.includes(s.id));
    if (!nextStage) return { newlyCompleted: [], advanced: false, progress: getProgress() };

    let passed = false;
    try {
      passed = nextStage.check(payload || {});
    } catch (e) {
      console.warn('[FlashStages] check error:', e);
    }

    if (passed) {
      completed.push(nextStage.id);
      saved.completed = completed;
      saved.updatedAt = new Date().toISOString();
      safeSet(STORAGE_KEY, saved);
      newlyCompleted.push(nextStage);

      // Notify caller
      if (onStageComplete) {
        try { onStageComplete(nextStage); } catch (e) {}
      }
    }

    return {
      newlyCompleted,
      advanced: newlyCompleted.length > 0,
      progress: getProgress()
    };
  }

# FLASH Backend — production Dockerfile
# Node 24 LTS + tsx runtime (no build step needed)

FROM node:24-alpine

# System deps
RUN apk add --no-cache tini

WORKDIR /app

# Install deps first (layer caching)
COPY package.json package-lock.json ./
RUN npm ci --omit=dev --ignore-scripts && \
    npm cache clean --force

# Source code + scripts
COPY src ./src
COPY scripts ./scripts
COPY tsconfig.json ./
COPY verify.ps1 ./

# Install tsx + typescript for runtime
RUN npm install --no-save tsx@4.19.2 typescript@5.6.3

# Run as non-root
RUN addgroup -S flash && adduser -S flash -G flash && chown -R flash:flash /app
USER flash

ENV NODE_ENV=production
ENV PORT=3000
ENV HOST=0.0.0.0

EXPOSE 3000

# Health check (liveness only — no DB)
HEALTHCHECK --interval=30s --timeout=5s --start-period=20s --retries=3 \
  CMD wget -q -O /dev/null http://localhost:3000/health || exit 1

# tini = proper signal handling
ENTRYPOINT ["/sbin/tini", "--"]

# Run directly via tsx
CMD ["npx", "tsx", "src/server.ts"]
